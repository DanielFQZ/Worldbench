cmake_minimum_required(VERSION 3.24)
project(Worldbench VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent ShaderTools Network)
qt_standard_project_setup(REQUIRES 6.8)
if(MSVC)
  add_compile_options(/utf-8 /permissive-)
  # Some CMake/Chinese MSVC combinations misdecode the /showIncludes probe.
  # A wrong prefix silently loses header dependencies in Ninja incremental builds.
  if(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
  endif()
endif()
set(KDDockWidgets_FRONTENDS qtquick CACHE STRING "" FORCE)
set(KDDockWidgets_STATIC ON CACHE BOOL "" FORCE)
set(KDDockWidgets_EXAMPLES OFF CACHE BOOL "" FORCE)
set(KDDockWidgets_TESTS OFF CACHE BOOL "" FORCE)
set(KDDockWidgets_LINTER OFF CACHE BOOL "" FORCE)
set(KDDockWidgets_QML_MODULE OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(kddockwidgets
  URL https://codeload.github.com/KDAB/KDDockWidgets/zip/66e66fa5d8a824488538558e274023822ac13952
  URL_HASH SHA256=ea9ad76d5c8b4c861c9c931b265f3a08713b0bc4d238fbc3185718f312b03c3a
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(kddockwidgets)
add_library(worldbench_core STATIC
  src/core/World.cpp src/core/HeightField.cpp src/core/Heightmap.cpp src/core/HeightmapRay.cpp src/core/TerrainProject.cpp src/core/Search.cpp src/core/Assets.cpp src/core/Project.cpp src/core/Mesh.cpp src/core/MinecraftExport.cpp)
target_include_directories(worldbench_core PUBLIC src)
target_link_libraries(worldbench_core PUBLIC Qt6::Core Qt6::Gui PRIVATE Qt6::CorePrivate)
qt_add_resources(worldbench_core block_names PREFIX /translations BASE src/core/resources FILES src/core/resources/zh_cn_blocks_1.20.1.json src/core/resources/pinyin_search.json)
qt_add_executable(Worldbench WIN32 src/main.cpp src/Editor.cpp src/EditorTools.cpp src/EditorTerrain.cpp src/McpServer.cpp src/Viewport.cpp src/ViewportGizmo.cpp app/resources.qrc)
target_link_libraries(Worldbench PRIVATE worldbench_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Network Qt6::GuiPrivate KDAB::kddockwidgets)
qt_add_shaders(Worldbench worldbench_shaders PREFIX /shaders BASE shaders FILES shaders/voxel.vert shaders/voxel.frag)
enable_testing()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
add_executable(worldbench_core_tests tests/CoreTests.cpp)
target_link_libraries(worldbench_core_tests PRIVATE worldbench_core)
add_test(NAME worldbench_core COMMAND worldbench_core_tests)
qt_add_executable(worldbench_ui_tests tests/UiTests.cpp src/Editor.cpp src/EditorTools.cpp src/EditorTerrain.cpp src/McpServer.cpp src/Viewport.cpp src/ViewportGizmo.cpp app/resources.qrc)
target_link_libraries(worldbench_ui_tests PRIVATE worldbench_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Network Qt6::GuiPrivate Qt6::Test KDAB::kddockwidgets)
qt_add_shaders(worldbench_ui_tests worldbench_test_shaders PREFIX /shaders BASE shaders FILES shaders/voxel.vert shaders/voxel.frag)
add_test(NAME worldbench_ui COMMAND worldbench_ui_tests)
set_tests_properties(worldbench_ui PROPERTIES TIMEOUT 120 LABELS "desktop")
add_executable(WorldbenchMcp src/McpBridge.cpp)
target_link_libraries(WorldbenchMcp PRIVATE Qt6::Core Qt6::Network)
add_executable(worldbench_asset_tests tests/AssetTests.cpp)
target_link_libraries(worldbench_asset_tests PRIVATE worldbench_core Qt6::CorePrivate)
add_test(NAME worldbench_assets COMMAND worldbench_asset_tests)
add_executable(worldbench_terrain_tests tests/TerrainTests.cpp)
target_link_libraries(worldbench_terrain_tests PRIVATE worldbench_core)
add_test(NAME worldbench_terrain COMMAND worldbench_terrain_tests)
